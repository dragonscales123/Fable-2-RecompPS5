# Fable 2 Recomp PS5: optional switches. Rename this file to ps5-env.txt in
# the title folder (/data/homebrew/PPSA99214/) to use it. A title gets no
# environment, so ps5_title.cpp reads KEY=VALUE lines from it before main.
#
# Any SDK setting (cvar) can be set as REX_<NAME>: gpu_vsync -> REX_GPU_VSYNC.
# Lines starting with # are ignored.

# Log every Vulkan debug message (slow; for chasing renderer problems).
#REX_VULKAN_LOG_DEBUG_MESSAGES=true

# Run a modified or non-catalogue default.xex (the hash is still logged).
#REX_SKIPHASH=true

# Sampling profiler: where each thread spends its time, every 10 s, in
# /app0/profile.txt (symbolize with symbols/fable_2.elf from the build).
#REX_PROFILE=1
#REX_PROFILE_HZ=250
#REX_PROFILE_INTERVAL=10

# RADV's GPU wait: microseconds to sleep between polls once its 1.5 ms spin
# is over (RADV itself sleeps 1000; 0 = just yield).
#FABLE2_RADV_WAIT_US=100

# Performance overlay (FPS, thread load, memory) at startup; L3 + R3 toggles it.
#REX_SHOW_DEBUG_OVERLAY=true

# Hot pages (pages the CPU keeps rewriting) are compared with their last upload
# once per guest CPU/GPU sync point instead of on every draw. If geometry or
# textures look stale or garbled, set this to false.
#REX_SHARED_MEMORY_HOT_PAGE_CHECK_EPOCHS=false
# Stricter: also restart the check at every GPU fence / memory write.
#REX_SHARED_MEMORY_HOT_PAGE_EPOCH_ON_GPU_SIGNALS=true

# Guest draw constants as dynamic uniform buffers (fewer descriptor writes per
# draw). If lighting, colours or object positions look wrong, set to false.
#REX_VULKAN_DYNAMIC_CONSTANT_BUFFERS=false

# A CPU write fault invalidates up to 1 MB around it, but stops at pages a
# texture watches. If textures look stale (not updating), set this to true.
#REX_SHARED_MEMORY_WIDEN_INTO_WATCHED_PAGES=true

# Frame cap. The game waits two vblanks per frame; the PS5 build ticks them at
# 120 Hz (2x the guest's 60 Hz), capping it at 60 fps. 1 = the original 30 fps
# cap. Leave REX_VSYNC on (unset).
#REX_GUEST_VBLANK_MULTIPLIER=1

# Submit GPU work every N draws (PS5 default 512; 0 = off), so a synchronous readback (memexport,
# needed by the glowing trail) has less queued work to wait for.
#REX_VULKAN_SUBMIT_EVERY_DRAWS=0

# Record and submit Vulkan command buffers on a separate thread (experimental).
#REX_VULKAN_SUBMIT_THREAD=true
